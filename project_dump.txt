import os
from launch import LaunchDescription
from launch.actions import IncludeLaunchDescription
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch_ros.actions import Node
from ament_index_python.packages import get_package_share_directory
import xacro

def generate_launch_description():
    pkg_maze_robot = get_package_share_directory('maze_robot')
    pkg_ros_gz_sim = get_package_share_directory('ros_gz_sim')

    # Process URDF
    xacro_file = os.path.join(pkg_maze_robot, 'urdf', 'maze_robot.urdf.xacro')
    doc = xacro.process_file(xacro_file)
    robot_desc = doc.toprettyxml(indent='  ')

    # Robot State Publisher (WITH use_sim_time: True to fix SLAM)
    robot_state_publisher_node = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        output='screen',
        parameters=[{
            'robot_description': robot_desc,
            'use_sim_time': True
        }]
    )

    # Gazebo
    world_file = os.path.join(pkg_maze_robot, 'worlds', 'maze_world.sdf')
    gazebo = IncludeLaunchDescription(
        PythonLaunchDescriptionSource(
            os.path.join(pkg_ros_gz_sim, 'launch', 'gz_sim.launch.py')
        ),
        launch_arguments={'gz_args': ['-r ', world_file]}.items()
    )

    # Spawn Robot
    spawn_node = Node(
        package='ros_gz_sim',
        executable='create',
        arguments=[
            '-name', 'maze_robot',
            '-topic', 'robot_description',
            '-x', '0.8',
            '-y', '11.8',
            '-z', '0.15'
        ],
        output='screen'
    )

    return LaunchDescription([
        robot_state_publisher_node,
        gazebo,
        spawn_node
    ])
